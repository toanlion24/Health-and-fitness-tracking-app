export const getNutritionData = async (req: Request, res: Response) => {
  const { queryText } = req.query; // Expecting things like "coke", "milk", "rice", etc.

  if (!queryText) {
    return res.status(400).json({ error: "Query text parameter is required." });
  }

  try {
    // ⚠️ Open Food Facts public global search endpoint (Returns JSON format)
    const url = `https://world.openfoodfacts.org/cgi/search.pl?search_terms=${encodeURIComponent(String(queryText))}&search_simple=1&action=process&json=1&page_size=5`;
    
    // Per documentation: Always supply a custom User-Agent so they don't block your student app!
    const response = await axios.get(url, {
      headers: {
        'User-Agent': 'StudentHealthApp - Android/iOS - Version 1.0'
      }
    });

    const products = response.data.products;

    if (!products || products.length === 0) {
      return res.status(404).json({ success: false, message: "No real food items matched your query." });
    }

    // Map the deeply nested Open Food Facts layout into the flat keys your mobile UI expects
    const safeData = products.map((prod: any) => ({
      name: prod.product_name || "Unknown Product",
      
      // ✅ 100% UNLOCKED AND FREE MACRONUTRIENTS (calculated per 100g)
      calories: prod.nutriments?.['energy-kcal_100g'] ?? 0,

      fat: prod.nutriments?.fat_100g ?? 0,
      servingSize: 100
    }));

    // Send the live data back to your React Native app!
    return res.status(200).json({ success: true, data: safeData });

  } catch (error: any) {
    console.error("Open Food Facts API connection error:", error.message);
